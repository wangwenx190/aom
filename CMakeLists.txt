cmake_minimum_required(VERSION 3.25)
project(AOM VERSION 3.15.3 LANGUAGES C)
include(GNUInstallDirs)
include(cmake/Sources.cmake)
find_package(Threads REQUIRED)

option(AOM_BUILD_STATIC "Build the static library" ON)
option(AOM_BUILD_SHARED "Build the shared library" ON)

set(aom_arch "${CMAKE_SYSTEM_PROCESSOR}")
if(MSVC AND CMAKE_C_COMPILER_ARCHITECTURE_ID)
    set(aom_arch "${CMAKE_C_COMPILER_ARCHITECTURE_ID}")
elseif(APPLE AND CMAKE_OSX_ARCHITECTURES)
    set(aom_arch "${CMAKE_OSX_ARCHITECTURES}")
endif()
string(TOLOWER "${aom_arch}" aom_arch)

set(aom_profile generic)
if(aom_arch MATCHES "^(arm64|aarch64)$")
    if(MSVC)
        # The upstream SVE source uses the AArch64 GNU-style compiler ABI.
        set(aom_profile arm64-neon)
    else()
        set(aom_profile arm64)
    endif()
elseif(aom_arch MATCHES "^(x86_64|amd64|x64)$")
    enable_language(ASM_NASM)
    set(aom_profile x86_64)
endif()
include(cmake/SimdSources.cmake)

if(UNIX)
    set(HAVE_UNISTD_H ON)
    set(HAVE_PTHREAD_H ON)
endif()
if(CMAKE_C_COMPILER_ID MATCHES "Clang|GNU")
    set(CONFIG_GCC ON)
endif()
configure_file(config/aom_config.h.in config/aom_config.h)
foreach(header IN ITEMS aom_dsp_rtcd.h aom_scale_rtcd.h av1_rtcd.h)
    configure_file("config/${aom_profile}/${header}" "config/${header}" COPYONLY)
endforeach()
if(aom_profile STREQUAL "x86_64")
    configure_file(config/x86_64/aom_config.asm config/aom_config.asm COPYONLY)
endif()
list(APPEND codec_sources config/aom_config.c)

add_library(aom_objects OBJECT ${codec_sources})
target_compile_features(aom_objects PRIVATE c_std_11)
target_include_directories(aom_objects PRIVATE "${CMAKE_CURRENT_BINARY_DIR}" .)
target_compile_definitions(aom_objects PRIVATE
    "$<$<COMPILE_LANGUAGE:C>:_FILE_OFFSET_BITS=64;_LARGEFILE_SOURCE>"
    "$<$<COMPILE_LANGUAGE:ASM_NASM>:PIC>")
set_target_properties(aom_objects PROPERTIES POSITION_INDEPENDENT_CODE ON)

function(add_aom target type)
    add_library(${target} ${type} $<TARGET_OBJECTS:aom_objects>)
    target_include_directories(${target} PUBLIC
        "$<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}>"
        "$<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>")
    target_link_libraries(${target} PRIVATE Threads::Threads)
    if(UNIX)
        target_link_libraries(${target} PRIVATE m)
    endif()
    if(AOM_INSTALL)
        install(TARGETS ${target})
    endif()
endfunction()

option(AOM_INSTALL "Install libaom" ON)
if(AOM_BUILD_STATIC)
    add_aom(aom STATIC)
    if(WIN32)
        set_target_properties(aom PROPERTIES OUTPUT_NAME aom_static)
    endif()
endif()
if(AOM_BUILD_SHARED)
    add_aom(aom_shared SHARED)
    set_target_properties(aom_shared PROPERTIES OUTPUT_NAME aom VERSION ${PROJECT_VERSION}
        SOVERSION 3 WINDOWS_EXPORT_ALL_SYMBOLS ON)
endif()
if(AOM_INSTALL)
    install(DIRECTORY aom/ DESTINATION "${CMAKE_INSTALL_INCLUDEDIR}/aom" FILES_MATCHING PATTERN "*.h")
    install(FILES LICENSE PATENTS DESTINATION "${CMAKE_INSTALL_DATADIR}/licenses/aom")
endif()

if(WIN32 AND TARGET aom_shared)
    enable_language(RC)
    configure_file(cmake/LibraryVersion.rc.in LibraryVersion.rc @ONLY)
    target_sources(aom_shared PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/LibraryVersion.rc")
endif()
